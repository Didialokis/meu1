Sub GerarFluxogramasCorporaHorizontais()
    Dim ws As Worksheet
    Dim metodos(1 To 5) As String
    Dim numEtapas(1 To 5) As Integer
    Dim etapas(1 To 5, 1 To 6) As String
    Dim i As Integer, j As Integer
    Dim shp As Shape
    Dim conn As Shape
    Dim topo As Single, esquerda As Single, largura As Single, altura As Single
    Dim espacamento As Single
    
    ' ==========================================
    ' DADOS DOS FLUXOGRAMAS (COM ETAPAS EXPANDIDAS)
    ' ==========================================
    
    ' Método 1
    metodos(1) = "M1_Curadoria_Manual"
    numEtapas(1) = 5
    etapas(1, 1) = "1. AQUISIÇÃO INICIAL:" & vbCrLf & "Coleta de dados (tuplas/benchmarks originais) ou redação manual por especialistas."
    etapas(1, 2) = "2. TRATAMENTO DE DADOS:" & vbCrLf & "Limpeza de ruídos, normalização de texto e formatação estruturada do dataset base."
    etapas(1, 3) = "3. ESTRUTURAÇÃO E TEMPLATES:" & vbCrLf & "Injeção determinística de variáveis do dataset em gabaritos de prompt."
    etapas(1, 4) = "4. PROCESSAMENTO:" & vbCrLf & "Revisão humana curada para garantir balanceamento de categorias e mitigação de viés."
    etapas(1, 5) = "5. RESULTADO FINAL:" & vbCrLf & "Corpus estruturado de alta pureza e precisão semântica para fine-tuning."
    
    ' Método 2
    metodos(2) = "M2_Self_Instruct"
    numEtapas(2) = 5
    etapas(2, 1) = "1. AQUISIÇÃO INICIAL (SEMENTE):" & vbCrLf & "Criação estrita de pequeno conjunto de tarefas exemplares de alta diversidade."
    etapas(2, 2) = "2. EXPANSÃO SINTÉTICA:" & vbCrLf & "LLM 'Professor' gera recursivamente milhares de instruções análogas às sementes."
    etapas(2, 3) = "3. GERAÇÃO DE RESPOSTAS:" & vbCrLf & "Execução de inferência no LLM para gerar as respostas detalhadas das novas instruções."
    etapas(2, 4) = "4. FILTRAGEM (ROUGE-L):" & vbCrLf & "Remoção automática de saídas duplicadas, mal formatadas ou de baixa diversidade."
    etapas(2, 5) = "5. RESULTADO FINAL:" & vbCrLf & "Corpus de instrução massivo focado em generalização de tarefas (General Assistant)."
    
    ' Método 3
    metodos(3) = "M3_Evol_Instruct"
    numEtapas(3) = 6
    etapas(3, 1) = "1. AQUISIÇÃO INICIAL:" & vbCrLf & "Obtenção de um corpus base contendo instruções diretas e perguntas simples."
    etapas(3, 2) = "2. MUTAÇÃO RECURSIVA:" & vbCrLf & "LLM aplica operadores evolutivos (restrições lógicas, ofuscação, passos múltiplos)."
    etapas(3, 3) = "3. FILTRAGEM DE MUTAÇÃO:" & vbCrLf & "Verificação e exclusão de mutações falhas ou que violaram as regras evolutivas."
    etapas(3, 4) = "4. RESOLUÇÃO COMPLEXA:" & vbCrLf & "LLM Professor processa e resolve as novas instruções de alta complexidade."
    etapas(3, 5) = "5. REFINAMENTO:" & vbCrLf & "Limpeza das respostas geradas e descarte sumário de alucinações lógicas/matemáticas."
    etapas(3, 6) = "6. RESULTADO FINAL:" & vbCrLf & "Corpus de altíssima complexidade analítica e exigência cognitiva."
    
    ' Método 4
    metodos(4) = "M4_Bootstrapping_STaR"
    numEtapas(4) = 5
    etapas(4, 1) = "1. AQUISIÇÃO INICIAL:" & vbCrLf & "Seleção de problemas base que possuem respostas/gabaritos exatos e verificáveis."
    etapas(4, 2) = "2. GERAÇÃO CoT (Chain-of-Thought):" & vbCrLf & "Modelo tenta resolver o problema documentando abertamente o passo-a-passo lógico."
    etapas(4, 3) = "3. EXTRAÇÃO DE RESPOSTA:" & vbCrLf & "Isolamento da resposta final (Parsing) dentro da longa trajetória gerada pelo modelo."
    etapas(4, 4) = "4. VALIDAÇÃO POR GABARITO (ORM):" & vbCrLf & "Filtro descarta totalmente as trajetórias cuja resposta extraída diverge do gabarito."
    etapas(4, 5) = "5. RESULTADO FINAL:" & vbCrLf & "Corpus de raciocínio puro, mitigado contra alucinações e justificativas espúrias."
    
    ' Método 5
    metodos(5) = "M5_Retrotraducao"
    numEtapas(5) = 6
    etapas(5, 1) = "1. AQUISIÇÃO INICIAL:" & vbCrLf & "Coleta massiva de textos humanos de alta qualidade na web (documentações, artigos)."
    etapas(5, 2) = "2. TRATAMENTO DE TEXTO BRUTO:" & vbCrLf & "Limpeza de HTML, deduplicação e remoção de conteúdos de baixa entropia."
    etapas(5, 3) = "3. INFERÊNCIA INVERSA:" & vbCrLf & "LLM lê o texto alvo e tenta deduzir: 'Qual instrução o usuário deveria ter dado para eu gerar isso?'."
    etapas(5, 4) = "4. FILTRAGEM SEMÂNTICA:" & vbCrLf & "Remoção de instruções genéricas que falham em capturar a especificidade do texto base."
    etapas(5, 5) = "5. AVALIAÇÃO (LLM-as-a-Judge):" & vbCrLf & "Segundo modelo pontua a aderência instrução/texto. O loop iterativo refina a geração."
    etapas(5, 6) = "6. RESULTADO FINAL:" & vbCrLf & "Corpus sintético que espelha respostas orgânicas e complexas redigidas por humanos."

    ' ==========================================
    ' MOTOR DE GERAÇÃO GRÁFICA NO EXCEL (MODERNO)
    ' ==========================================
    Application.ScreenUpdating = False
    
    For i = 1 To 5
        ' Cria nova aba no final
        Set ws = ThisWorkbook.Sheets.Add(After:=ThisWorkbook.Sheets(ThisWorkbook.Sheets.Count))
        ws.Name = metodos(i)
        
        ' Remove linhas de grade para visual mais limpo
        ws.Activate
        ActiveWindow.DisplayGridlines = False
        
        ' Adiciona o Título da Página
        ws.Cells(2, 2).Value = "Fluxograma Metodológico: " & Replace(metodos(i), "_", " ")
        ws.Cells(2, 2).Font.Size = 16
        ws.Cells(2, 2).Font.Bold = True
        ws.Cells(2, 2).Font.Color = RGB(31, 73, 125)
        
        ' Configurações iniciais de tamanho e layout
        topo = 80
        esquerda = 30
        largura = 190    ' Reduzido levemente para caber confortavelmente 6 etapas na tela
        altura = 110
        espacamento = 40 ' Distância reduzida proporcionalmente
        
        For j = 1 To numEtapas(i)
            ' Adiciona caixa com cantos arredondados (Design Moderno)
            Set shp = ws.Shapes.AddShape(msoShapeRoundedRectangle, esquerda, topo, largura, altura)
            shp.Adjustments.Item(1) = 0.15 ' Define a curvatura dos cantos
            
            ' Aplica sombreamento suave
            shp.Shadow.Visible = msoTrue
            shp.Shadow.Type = msoShadow21
            shp.Shadow.Transparency = 0.6
            shp.Shadow.Size = 100
            shp.Shadow.Blur = 8
            shp.Shadow.OffsetX = 2
            shp.Shadow.OffsetY = 2
            
            ' Configurações de Texto
            With shp.TextFrame2
                .TextRange.Text = etapas(i, j)
                .VerticalAnchor = msoAnchorMiddle
                .TextRange.ParagraphFormat.Alignment = msoAlignCenter
                .TextRange.Font.Size = 10
                .TextRange.Font.Name = "Segoe UI" ' Fonte mais moderna
                .TextRange.Font.Bold = True
                .TextRange.Font.Fill.ForeColor.RGB = RGB(255, 255, 255) ' Letra branca
            End With
            
            ' Configurações de Cores das Fases (Paleta Refinada)
            If j = 1 Then
                shp.Fill.ForeColor.RGB = RGB(46, 117, 182) ' Azul Corporativo (Início)
            ElseIf j = numEtapas(i) Then
                shp.Fill.ForeColor.RGB = RGB(67, 160, 71) ' Verde Vivo (Final)
            Else
                shp.Fill.ForeColor.RGB = RGB(96, 125, 139) ' Cinza-Azulado Metálico (Etapas Intermediárias)
            End If
            
            ' Remove a borda padrão para um visual mais limpo
            shp.Line.Visible = msoFalse
            
            ' Adiciona seta apontando para a caixa à direita (exceto na última)
            If j < numEtapas(i) Then
                Set conn = ws.Shapes.AddConnector(msoConnectorStraight, esquerda + largura, topo + (altura / 2), esquerda + largura + espacamento, topo + (altura / 2))
                conn.Line.EndArrowheadStyle = msoArrowheadTriangle
                conn.Line.EndArrowheadWidth = msoArrowheadWidthMedium
                conn.Line.EndArrowheadLength = msoArrowheadLengthMedium
                conn.Line.Weight = 2.5
                conn.Line.ForeColor.RGB = RGB(176, 190, 197) ' Cinza Claro/Azulado para a Seta
            End If
            
            ' Desloca o eixo X para a próxima etapa
            esquerda = esquerda + largura + espacamento
        Next j
    Next i
    
    Application.ScreenUpdating = True
    MsgBox "Os fluxogramas com design moderno e etapas expandidas foram gerados com sucesso!", vbInformation, "Engenharia de Corpora"
End Sub
