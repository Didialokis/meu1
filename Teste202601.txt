import time
import boto3
from langchain_aws import ChatBedrock
from crewai import Agent, Task, Crew, Process

# ==========================================
# 1. CONEXÃO COM O AWS BEDROCK (LLAMA 3)
# ==========================================
print("Conectando ao Llama 3 via AWS Bedrock...")

# Usando o Llama 3 8B nativo da AWS. 
# Se preferir o 70B, troque para: 'meta.llama3-70b-instruct-v1:0'
bedrock_client = boto3.client(service_name="bedrock-runtime", region_name="us-east-1")

llm_bedrock = ChatBedrock(
    client=bedrock_client,
    model_id="meta.llama3-8b-instruct-v1:0",
    model_kwargs={"temperature": 0.0} # Mantém o determinismo
)

# ==========================================
# 2. DEFINIÇÃO DA TAREFA DE NEGÓCIOS
# ==========================================
business_task = """
Analise a viabilidade logística e financeira de importar lotes de notebooks refurbhished 
(com defeitos de placa-mãe para reparo) através de um hub logístico em Madri, Espanha, 
para distribuição no Brasil. Considere os custos de frete, tempo de trânsito alfandegário 
e o risco de avarias em componentes sensíveis como chips BGA durante o transporte.
Forneça um parecer final de recomendação (Aprovado ou Reprovado) com 3 justificativas.
"""

# ==========================================
# 3. TESTE 1: ARQUITETURA MONOLÍTICA
# ==========================================
print("\n--- INICIANDO TESTE MONOLÍTICO ---")
start_time_mono = time.time()

mono_agent = Agent(
    role='Analista Sênior de Operações',
    goal='Resolver o problema logístico e financeiro de ponta a ponta',
    backstory='Você é um especialista generalista capaz de analisar operações complexas.',
    llm=llm_bedrock,
    verbose=False,
    allow_delegation=False
)

mono_task = Task(
    description=f'Resolva a seguinte tarefa: {business_task}',
    expected_output='Parecer final (Aprovado/Reprovado) com 3 justificativas.',
    agent=mono_agent
)

mono_crew = Crew(agents=[mono_agent], tasks=[mono_task])

# Executa e captura o uso de tokens nativo do CrewAI/LangChain
resultado_mono = mono_crew.kickoff()

end_time_mono = time.time()
latencia_mono = end_time_mono - start_time_mono
usage_mono = resultado_mono.token_usage

print("Resultado Monolítico gerado.")

# ==========================================
# 4. TESTE 2: ARQUITETURA MULTI-AGENTE
# ==========================================
print("\n--- INICIANDO TESTE MULTI-AGENTE ---")
start_time_multi = time.time()

logistics_agent = Agent(
    role='Especialista em Logística Internacional',
    goal='Analisar rotas de importação e riscos de transporte de hardware sensível',
    backstory='Você é um engenheiro especialista em gerenciar o trânsito de eletrônicos entre a Europa e o Brasil.',
    llm=llm_bedrock, 
    verbose=False,
    allow_delegation=False
)

financial_agent = Agent(
    role='Auditor de Viabilidade Financeira',
    goal='Avaliar o parecer logístico e emitir a recomendação final de viabilidade',
    backstory='Você é um auditor financeiro rigoroso que só aprova operações com risco mitigado.',
    llm=llm_bedrock,
    verbose=False,
    allow_delegation=False
)

task_1 = Task(
    description=f'Analise os riscos logísticos do seguinte cenário: {business_task}',
    expected_output='Um relatório de riscos operacionais e estimativa de avarias de hardware.',
    agent=logistics_agent
)

task_2 = Task(
    description='Com base no relatório logístico, emita o parecer final (Aprovado/Reprovado) com 3 justificativas.',
    expected_output='Parecer final de negócios estruturado em tópicos.',
    agent=financial_agent
)

multi_crew = Crew(
    agents=[logistics_agent, financial_agent],
    tasks=[task_1, task_2],
    process=Process.sequential
)

resultado_multi = multi_crew.kickoff()

end_time_multi = time.time()
latencia_multi = end_time_multi - start_time_multi
usage_multi = resultado_multi.token_usage

# ==========================================
# 5. RESULTADOS DO BENCHMARK
# ==========================================
print("\n==========================================")
print(" RESULTADOS PRELIMINARES DO EXPERIMENTO")
print("==========================================")

print("\n[ARQUITETURA MONOLÍTICA]")
print(f"Latência: {latencia_mono:.2f} segundos")
if usage_mono:
    print(f"Tokens Totais: {usage_mono.total_tokens} (Prompt: {usage_mono.prompt_tokens}, Completion: {usage_mono.completion_tokens})")

print("\n[ARQUITETURA MULTI-AGENTE]")
print(f"Latência: {latencia_multi:.2f} segundos")
if usage_multi:
    print(f"Tokens Totais: {usage_multi.total_tokens} (Prompt: {usage_multi.prompt_tokens}, Completion: {usage_multi.completion_tokens})")
print("==========================================")
