import time
import torch
from transformers import AutoModelForCausalLM, AutoTokenizer, pipeline, BitsAndBytesConfig
from langchain_core.language_models.llms import LLM as LangChainLLM
from pydantic import PrivateAttr
from typing import Any, List, Optional
from crewai import Agent, Task, Crew, Process

# ==========================================
# 0. CLASSE WRAPPER (A Solução do Erro)
# ==========================================
class CustomLocalLLM(LangChainLLM):
    """
    Wrapper customizado para ignorar os conflitos de validação do Pydantic no CrewAI.
    Herdamos diretamente do BaseLLM do LangChain.
    """
    _pipeline_func: Any = PrivateAttr()

    def __init__(self, pipeline_func, **kwargs):
        super().__init__(**kwargs)
        self._pipeline_func = pipeline_func

    def _call(self, prompt: str, stop: Optional[List[str]] = None, **kwargs) -> str:
        # O HuggingFace pipeline com return_full_text=False já devolve apenas a resposta gerada
        resposta = self._pipeline_func(prompt)
        return resposta[0]["generated_text"]

    @property
    def _llm_type(self) -> str:
        return "custom_local_llama"

# ==========================================
# 1. CONFIGURAÇÃO DO MODELO LOCAL (AWS)
# ==========================================
model_id = "meta-llama/Meta-Llama-3.1-8B-Instruct"

print("Carregando o modelo na GPU...")
tokenizer = AutoTokenizer.from_pretrained(model_id)

# Nova forma de configurar a quantização para as GPUs da AWS
quant_config = BitsAndBytesConfig(
    load_in_4bit=True,
    bnb_4bit_compute_dtype=torch.float16,
    bnb_4bit_quant_type="nf4",
    bnb_4bit_use_double_quant=True
)

model = AutoModelForCausalLM.from_pretrained(
    model_id,
    device_map="auto",
    quantization_config=quant_config,
)

model.config.pad_token_id = tokenizer.eos_token_id

text_pipeline = pipeline(
    "text-generation",
    model=model,
    tokenizer=tokenizer,
    max_new_tokens=1024,
    do_sample=False, # Garante determinismo sem conflitar com temperature
    return_full_text=False,
    pad_token_id=tokenizer.eos_token_id
)

# Instanciando o nosso LLM customizado em vez do ChatHuggingFace problemático
local_llm = CustomLocalLLM(pipeline_func=text_pipeline)

# ==========================================
# 2. DEFINIÇÃO DA TAREFA DE NEGÓCIOS
# ==========================================
business_task = """
Analise a viabilidade logística e financeira de importar lotes de notebooks refurbhished 
(com defeitos de placa-mãe para reparo) através de um hub logístico em Madri, Espanha, 
para distribuição no Brasil. Considere os custos de frete, tempo de trânsito alfandegário 
e o risco de avarias em componentes sensíveis como chips BGA durante o transporte.
Forneça um parecer final de recomendação (Aprovado ou Reprovado) com 3 justificativas.
"""

# ==========================================
# 3. TESTE 1: ARQUITETURA MONOLÍTICA
# ==========================================
print("\n--- INICIANDO TESTE MONOLÍTICO ---")
start_time_mono = time.time()

prompt_monolitico = f"Você é um analista sênior de operações. Resolva a seguinte tarefa: {business_task}"

# Chamada direta usando o nosso LLM customizado
resposta_mono = local_llm.invoke(prompt_monolitico)

end_time_mono = time.time()
latencia_mono = end_time_mono - start_time_mono

# Contagem de tokens (o invoke de LangChainLLM devolve a string diretamente)
tokens_in_mono = len(tokenizer.encode(prompt_monolitico))
tokens_out_mono = len(tokenizer.encode(resposta_mono))
total_tokens_mono = tokens_in_mono + tokens_out_mono

print(f"Resultado Monolítico gerado com sucesso.")

# ==========================================
# 4. TESTE 2: ARQUITETURA MULTI-AGENTE (CrewAI)
# ==========================================
print("\n--- INICIANDO TESTE MULTI-AGENTE ---")
start_time_multi = time.time()

# Agentes agora recebem o local_llm envelopado corretamente
logistics_agent = Agent(
    role='Especialista em Logística Internacional',
    goal='Analisar rotas de importação e riscos de transporte de hardware',
    backstory='Você é especialista em gerenciar o trânsito de eletrônicos sensíveis entre a Europa e o Brasil.',
    llm=local_llm, 
    verbose=False,
    allow_delegation=False
)

financial_agent = Agent(
    role='Auditor de Viabilidade Financeira',
    goal='Avaliar o parecer logístico e emitir a recomendação final de viabilidade',
    backstory='Você é um auditor rigoroso que só aprova operações com risco mitigado e alta margem de retorno.',
    llm=local_llm,
    verbose=False,
    allow_delegation=False
)

task_1 = Task(
    description=f'Analise os riscos logísticos do seguinte cenário: {business_task}',
    expected_output='Um relatório de riscos operacionais e estimativa de avarias.',
    agent=logistics_agent
)

task_2 = Task(
    description='Com base no relatório logístico, emita o parecer final (Aprovado/Reprovado) com 3 justificativas.',
    expected_output='Parecer final estruturado em tópicos.',
    agent=financial_agent
)

crew = Crew(
    agents=[logistics_agent, financial_agent],
    tasks=[task_1, task_2],
    process=Process.sequential
)

resposta_multi = crew.kickoff()

end_time_multi = time.time()
latencia_multi = end_time_multi - start_time_multi

tokens_in_multi = len(tokenizer.encode(business_task)) * 3 
tokens_out_multi = len(tokenizer.encode(resposta_multi.raw))
total_tokens_multi = tokens_in_multi + tokens_out_multi

# ==========================================
# 5. RESULTADOS DO BENCHMARK
# ==========================================
print("\n==========================================")
print(" RESULTADOS PRELIMINARES DO EXPERIMENTO")
print("==========================================")
print("\n[ARQUITETURA MONOLÍTICA]")
print(f"Latência: {latencia_mono:.2f} segundos")
print(f"Tokens Consumidos (estimado): {total_tokens_mono}")

print("\n[ARQUITETURA MULTI-AGENTE]")
print(f"Latência: {latencia_multi:.2f} segundos")
print(f"Tokens Consumidos (estimado): > {total_tokens_multi}")
print("==========================================")
