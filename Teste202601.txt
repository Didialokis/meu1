import time
from langchain_aws import ChatBedrock
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.messages import SystemMessage, HumanMessage

# ==========================================
# 1. CONFIGURAÇÃO DO MODELO (AMAZON BEDROCK)
# ==========================================
print("Conectando ao Amazon Bedrock...")

# COLE AQUI A SUA CONFIGURAÇÃO DO BOTO3 E CHATBEDROCK QUE ESTAVA FUNCIONANDO
# Exemplo (mantenha a sua configuração de client e profile):
import boto3
boto_session = boto3.Session() # Ajuste se você usa profile específico
bedrock_client = boto_session.client(service_name='bedrock-runtime')

llm = ChatBedrock(
    client=bedrock_client,
    model_id="meta.llama3-8b-instruct-v1:0", # Ajuste para o ID exato que você tem acesso no Bedrock (ex: anthropic.claude-3-sonnet...)
    model_kwargs={"temperature": 0.0}
)

# ==========================================
# 2. DEFINIÇÃO DA TAREFA DE NEGÓCIOS
# ==========================================
business_task = """
Analise a viabilidade logística e financeira de importar lotes de notebooks refurbhished 
(com defeitos de placa-mãe para reparo) através de um hub logístico em Madri, Espanha, 
para distribuição no Brasil. Considere os custos de frete, tempo de trânsito alfandegário 
e o risco de avarias em componentes sensíveis como chips BGA durante o transporte.
Forneça um parecer final de recomendação (Aprovado ou Reprovado) com 3 justificativas.
"""

# ==========================================
# 3. TESTE 1: ARQUITETURA MONOLÍTICA
# ==========================================
print("\n--- INICIANDO TESTE MONOLÍTICO ---")
start_time_mono = time.time()

mensagens_mono = [
    SystemMessage(content="Você é um analista sênior de operações corporativas. Resolva problemas complexos de ponta a ponta com raciocínio lógico estruturado."),
    HumanMessage(content=f"Resolva a seguinte tarefa passo a passo:\n{business_task}")
]

resposta_mono = llm.invoke(mensagens_mono)

end_time_mono = time.time()
latencia_mono = end_time_mono - start_time_mono

# Como o Bedrock já devolve o metadata com o uso de tokens, não precisamos de tokenizer!
tokens_in_mono = resposta_mono.response_metadata.get('usage', {}).get('prompt_tokens', 0)
tokens_out_mono = resposta_mono.response_metadata.get('usage', {}).get('completion_tokens', 0)
total_tokens_mono = tokens_in_mono + tokens_out_mono

print("Resultado Monolítico gerado com sucesso.")

# ==========================================
# 4. TESTE 2: ARQUITETURA MULTI-AGENTE (Orquestração Nativa)
# ==========================================
print("\n--- INICIANDO TESTE MULTI-AGENTE ---")
start_time_multi = time.time()

# --- Agente 1: Logística ---
mensagens_agente1 = [
    SystemMessage(content="Você é um Engenheiro Especialista em Logística Internacional."),
    HumanMessage(content=f"Analise EXCLUSIVAMENTE os riscos logísticos operacionais e a estimativa de avarias do seguinte cenário:\n{business_task}")
]
resposta_agente1 = llm.invoke(mensagens_agente1)
texto_agente1 = resposta_agente1.content

# --- Agente 2: Finanças (Recebe o Output do Agente 1) ---
mensagens_agente2 = [
    SystemMessage(content="Você é um Auditor de Viabilidade Financeira rigoroso. Você só aprova operações com base em relatórios logísticos prévios."),
    HumanMessage(content=f"Com base no Relatório Logístico abaixo, emita o parecer final financeiro (Aprovado/Reprovado) com 3 justificativas.\n\nRelatório Logístico Recebido:\n{texto_agente1}")
]
resposta_agente2 = llm.invoke(mensagens_agente2)

end_time_multi = time.time()
latencia_multi = end_time_multi - start_time_multi

# Cálculo preciso de tokens na arquitetura Multi-Agente somando as chamadas
tokens_in_ag1 = resposta_agente1.response_metadata.get('usage', {}).get('prompt_tokens', 0)
tokens_out_ag1 = resposta_agente1.response_metadata.get('usage', {}).get('completion_tokens', 0)

tokens_in_ag2 = resposta_agente2.response_metadata.get('usage', {}).get('prompt_tokens', 0)
tokens_out_ag2 = resposta_agente2.response_metadata.get('usage', {}).get('completion_tokens', 0)

total_tokens_multi = tokens_in_ag1 + tokens_out_ag1 + tokens_in_ag2 + tokens_out_ag2

# ==========================================
# 5. RESULTADOS DO BENCHMARK
# ==========================================
print("\n==========================================")
print(" RESULTADOS PRELIMINARES DO EXPERIMENTO")
print("==========================================")
print("\n[ARQUITETURA MONOLÍTICA]")
print(f"Latência: {latencia_mono:.2f} segundos")
print(f"Tokens Consumidos: {total_tokens_mono}")

print("\n[ARQUITETURA MULTI-AGENTE]")
print(f"Latência: {latencia_multi:.2f} segundos")
print(f"Tokens Consumidos: {total_tokens_multi}")
print("==========================================")
